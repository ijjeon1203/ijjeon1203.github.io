푸쉬 과정
git add .	파일들 추가
git commit -m "ㅇㅇㅇ"	파일들 추가 후 커밋
git push -u origin 브랜치	해당 브랜치에 푸쉬

git commit -am "ㅇㅇㅇ"	add와 commit을 동시에 해줌

원격저장소 연결
git remote add origin 주소	원격저장소 추가
git remote -v		원격저장소 목록 확인

파일 이름 변경
git mv 지금이름 바꿀이름	파일 이름 변경
git rm 바꾸기전이름	파일 바꾸기 전 이름 삭제
git add 바꾼이름		파일 바꾼 이름 추가

branch
git branch 브랜치		브랜치 생성
git checkout 브랜치	현재 있는 브랜치에서 다른 브랜치로 변경
git checkout -b 브랜치 	브랜치 생성과 동시에 바로 변경
git branch -d 브랜치	브랜치 삭제

git push origin --all	원격저장소에 모든 브랜치 전달

git 파일 삭제(파일명을 한칸씩 띄어서 쓰면 여러개 파일 한번에 삭제 가능)
git rm -r --cached 폴더	원격저장소 파일 삭제 로컬은 남아있음
git commit -m ""		삭제 후 커밋
git push origin 브랜치	원격에 저장

git 파일 삭제 후
git add -A
git stash

git fetch 			로컬에는 없지만 원격에는 있는 데이터를 가져옴

git clone 주소		원격저장소에 있는 데이터를 로컬로 복사
git fork 주소		다른사람 원격저장소를 내 원격저장소로 복사

git pull origin 브랜치	특정 브랜치만 데이터 로컬로 가져오기

git status 		git 상태 보기
